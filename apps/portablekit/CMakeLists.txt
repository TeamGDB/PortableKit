cmake_minimum_required(VERSION 3.20)
# PortableKit as one desktop program: the player brings the disc image, the
# program identifies the game, plays it under the interpreter at once and
# recompiles it on the player's machine meanwhile. See docs/DESKTOP_APP.md.
#
#   cmake -S apps/portablekit -B out-app -G Ninja -DCMAKE_BUILD_TYPE=Release
#   cmake --build out-app -j2
#   out-app/bin/portablekit --help
#
# It ships no game code and no keys: it links no generated code
# (portablekit_add_game NO_CORPUS) and no built-in keys (EXTERNAL_KEYS).
project(PortableKitApp VERSION 0.1.0 LANGUAGES CXX)

# The framework's tests are built by the framework's own build.
set(PSPRECOMP_BUILD_TESTS OFF CACHE BOOL "Build reusable PSPRecomp tests")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../.." portablekit)

# Hand-written profiles to carry as optional extras, as paths to each port's
# profile source (e.g. .../Purun/host/purun_profile.cpp). Each defines
# portablekit::game(); it is compiled under its own name and chosen when the
# player's executable matches its hash.
set(PORTABLEKIT_APP_PROFILES "" CACHE STRING "Profile sources of ports to include as optional extras")

set(app_sources
    src/app_main.cpp
    src/app_home.cpp
    src/auto_profile.cpp
    src/compile_control.cpp
    src/corpus.cpp
    src/corpus_loader.cpp
    src/game_import.cpp
    src/keys_file.cpp
    src/launcher_ui.cpp
    src/process.cpp
    src/response_file.cpp
    src/text_format.cpp)

set(profile_index 0)
set(profile_entries "")
set(profile_declarations "")
foreach(profile_source IN LISTS PORTABLEKIT_APP_PROFILES)
    get_filename_component(profile_source "${profile_source}" ABSOLUTE)
    get_filename_component(profile_name "${profile_source}" NAME_WE)
    set(function_name "portablekit_app_profile_${profile_index}")
    # The profile's own definition of game() becomes this function; profile.hpp
    # is renamed the same way in that file, so its hooks still find it.
    set_property(SOURCE "${profile_source}" APPEND PROPERTY COMPILE_DEFINITIONS "game=${function_name}")
    list(APPEND app_sources "${profile_source}")
    string(APPEND profile_declarations "const GameProfile &${function_name}();\n")
    string(APPEND profile_entries "    {\"${profile_name}\", &${function_name}},\n")
    math(EXPR profile_index "${profile_index} + 1")
endforeach()
set(profiles_inc "${CMAKE_CURRENT_BINARY_DIR}/generated_app/hand_profiles.inc")
file(CONFIGURE OUTPUT "${profiles_inc}" CONTENT
"// Generated by apps/portablekit/CMakeLists.txt from PORTABLEKIT_APP_PROFILES.
namespace portablekit {
${profile_declarations}}
#define PORTABLEKIT_APP_HAND_PROFILES \\
${profile_entries}
")

portablekit_add_game(portablekit
    PROFILE_DIR "${CMAKE_CURRENT_SOURCE_DIR}"
    SOURCES ${app_sources}
    EXTERNAL_KEYS
    NO_CORPUS
    HOST_MAIN_NAME portablekit_host_main)

# The corpus ABI: what a library compiled on the player's machine must have
# been compiled against to be loaded by this build. It hashes every header the
# generated code includes and every source of the recompiler, so changing any
# of them invalidates every cache (docs/DESKTOP_APP.md, "Cache").
set(abi_header "${CMAKE_CURRENT_BINARY_DIR}/generated_app/corpus_abi.hpp")
# Generated code includes corpus_abi.hpp (and through it allegrex_context.hpp)
# and nothing else of the framework; what it contains is decided by the
# recompiler and the analysis and decoding it uses.
set(abi_inputs
    "${PORTABLEKIT_ROOT}/include/psprecomp/corpus_abi.hpp"
    "${PORTABLEKIT_ROOT}/include/psprecomp/allegrex_context.hpp"
    "${PORTABLEKIT_ROOT}/include/psprecomp/decoder.hpp"
    "${PORTABLEKIT_ROOT}/include/psprecomp/program_analysis.hpp"
    "${PORTABLEKIT_ROOT}/include/psprecomp/elf32.hpp"
    "${PORTABLEKIT_ROOT}/tools/codegen_main.cpp"
    "${PORTABLEKIT_ROOT}/src/decoder.cpp"
    "${PORTABLEKIT_ROOT}/src/program_analysis.cpp"
    "${PORTABLEKIT_ROOT}/src/elf32.cpp")
get_property(abi_definitions GLOBAL PROPERTY PORTABLEKIT_COMPILE_DEFINITIONS)
string(REPLACE ";" "|" abi_inputs_arg "${abi_inputs}")
string(REPLACE ";" "|" abi_definitions_arg "${abi_definitions}")
add_custom_command(
    OUTPUT "${abi_header}"
    COMMAND ${CMAKE_COMMAND} "-DINPUTS=${abi_inputs_arg}" "-DDEFINITIONS=${abi_definitions_arg}"
            "-DOUTPUT=${abi_header}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/corpus_abi.cmake"
    DEPENDS ${abi_inputs} "${CMAKE_CURRENT_SOURCE_DIR}/cmake/corpus_abi.cmake"
    COMMENT "Computing the corpus ABI"
    VERBATIM)
target_sources(portablekit PRIVATE "${abi_header}")
target_include_directories(portablekit PRIVATE
    "${CMAKE_CURRENT_BINARY_DIR}/generated_app" "${CMAKE_CURRENT_SOURCE_DIR}/src")

# What the program needs beside it to compile a game: the recompiler and the
# headers the generated code includes. A release bundles both (on macOS inside
# the .app); here they are copied next to the executable.
add_dependencies(portablekit psp_recomp)
add_custom_command(TARGET portablekit POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:portablekit>/share/portablekit/include"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${PORTABLEKIT_ROOT}/include/psprecomp"
            "$<TARGET_FILE_DIR:portablekit>/share/portablekit/include/psprecomp"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:psp_recomp>" "$<TARGET_FILE_DIR:portablekit>/"
    COMMENT "Placing the recompiler and its headers next to portablekit")
